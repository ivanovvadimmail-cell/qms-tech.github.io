async function decryptSite(password) { const qmsUrl = new URL( "qms.dat", document.baseURI ).href; console.log("Загрузка:", qmsUrl); const response = await fetch( qmsUrl, { cache: "no-store" } ); if (!response.ok) { throw new Error( "Не удалось загрузить qms.dat. HTTP " + response.status + "\n" + qmsUrl ); } const data = new Uint8Array( await response.arrayBuffer() ); const magic = new TextDecoder().decode( data.slice(0, 4) ); if (magic !== "QMS1") { throw new Error( "qms.dat имеет неправильный формат." ); } const salt = data.slice(4, 20); const iv = data.slice(20, 32); const encrypted = data.slice(32); const key = await deriveKey( password, salt ); let decrypted; try { decrypted = new Uint8Array( await crypto.subtle.decrypt( { name: "AES-GCM", iv: iv }, key, encrypted ) ); } catch (error) { throw new Error( "Неверный ключ доступа." ); } const zip = fflate.unzipSync( decrypted ); files = zip; createBlobUrls(); processAllCss(); unlocked = true; history.replaceState( {}, "", window.location.pathname ); await renderCurrentPage(); }
